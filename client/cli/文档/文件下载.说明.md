# 下载文件喵()

为了统一下载入口

```js
//必要引入
import { 下载文件喵 } from "../工具/文件下载.js";
```

**函数展示**

```js
下载文件喵({下载url, 下载路径, 请求头, 任务列表, 进度条, 解析函数, 写入流, 转换, 并发数, 间隔})

//任务列表模式
- 下载{目标url, 下载路径, 文件名称}
- 解析{目标url, 文件名称}
```

**参数说明**

| 参数 | 类型 | 模式限制 | 说明 |
|------|------|------|------|
| 下载url | 字符串 | 流 | 下载链接 |
| 下载路径 | 字符串 | 流 | 下载路径，一般不用管 |
| 请求头 | 对象 |  | HTTP 请求附带的额外信息，一般不需要 |
| 任务列表 | 对象 |  | url集合 |
| 进度条 | 对象 |  | 控制进度条显示什么 |
| 解析函数 | 对象 |  | 告诉上层函数方法 |
| 写入流 | 对象 |  | 控制写入目标 |
| 转换 | 对象 |  | 选择是否转简繁 |
| 并发数 | 数值 |  | 指定多少个工作者同时工作，默认5 |
| 间隔 | 数值 |  | 执行每个任务的间隔，默认500ms |

**参数用法**

```js
//下载url 
await 下载文件喵(下载url, ...);
下载url = https://...

//下载路径
await 下载文件喵(..., 下载路径, ...);
下载路径 = .../....txt
//萝莉/妈妈.txt
//可以选择调用 清理文件名() 格式化文本以避免非法路径问题

//请求头

```

## 用法

### 单文件 - 流

```js
await 下载文件喵(url, 文件路径, { 请求头 });
```

---

### 多文件 - 池

```js
const 进度条 = new 进度({ 显示: ["百分比", "卷进度", "速度", "剩余"] });
进度条.设置总量(任务列表.length);

await 下载文件喵(null, null, {
  任务列表: [
    { url: "下载url 1", 路径: "/下载/小说/？？？.txt", 名称: "？？？" },
    { url: "下载url 2", 路径: "/下载/小说/？？？.txt", 名称: "？？？" },
    ...
  ],
  请求头: { Referer: "..." },
  进度条,
  并发数: 5, //推荐值
  间隔: 500, //推荐值
});

进度条.完成();
```

---

### 多解析 - 池

```js
import { 创建写入流 } from "../工具/文件写入.js";
import { 进度 } from "../工具/下载进度.js";

const 写入流 = 创建写入流("/下载/...");
const 进度条 = new 进度({ 显示: ["百分比", "章进度", "章速", "体积", "速度", "剩余"] });
进度条.设置总量(任务列表.length);

await 下载文件喵(null, null, {
  任务列表: [
    { url: "解析目标1", 名称: "？？？" },
    { url: "解析目标2", 名称: "？？？" },
    ...
  ],
  解析函数: (body) => ({ 标题: "...", 正文: "..." }),
  写入流,
  转换: (text) => text,
  进度条,
  并发数: 5, //推荐值
  间隔: 500, //推荐值
});

进度条.完成();
写入流.结束();
```

---

## 内部流程

```js
下载文件喵(url, 路径)
  └─ 流式: 发送请求 → 写入流 → 进度条 → 完成

下载文件喵(null, null, { 任务列表, 解析函数, ... })
  └─ 批量-解析: 池(N工作者) → 发送请求 → 解析 → 写入流 → 进度条

下载文件喵(null, null, { 任务列表, ... })
  └─ 批量-文件: 池(N工作者) → 发送请求 → 流式写文件 → 进度条
```
